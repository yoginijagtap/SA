#!/bin/bash
# --------------------------------------------
# Versioning strategy for WAR file in Jenkins
# Format: projectname-build_id-timestamp.war
# --------------------------------------------

# Exit on error
set -e

# Variables
PROJECT_NAME="vprofile-project"
TIMESTAMP=$(date +%Y%m%d-%H%M%S)
VERSIONS_DIR="$WORKSPACE/versions"

# Create versions directory if not exists
mkdir -p "$VERSIONS_DIR"

# Locate the WAR file (built by Maven earlier)
WAR_FILE=$(find "$WORKSPACE" -type f -name "*.war" | head -n 1)

if [ -z "$WAR_FILE" ]; then
  echo " ERROR: No WAR file found in workspace!"
  exit 1
fi

# Create versioned WAR filename
VERSIONED_WAR="${PROJECT_NAME}-${BUILD_ID}-${TIMESTAMP}.war"

# Copy and rename WAR file into versions directory
cp "$WAR_FILE" "$VERSIONS_DIR/$VERSIONED_WAR"

echo " WAR file versioned and stored successfully:"
echo "$VERSIONS_DIR/$VERSIONED_WAR"
