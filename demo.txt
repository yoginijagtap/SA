// Implements a thread-safe Singleton LoggerService

public class LoggerService {

    private static volatile LoggerService instance;

    private LoggerService() {}

    public static LoggerService getInstance() {
        if (instance == null) {
            synchronized (LoggerService.class) {
                if (instance == null) {
                    instance = new LoggerService();
                }
            }
        }
        return instance;
    }

    public void log(String message) {
        System.out.println("[LOG] " + message);
    }

    public static void main(String[] args) {
        LoggerService logger1 = LoggerService.getInstance();
        LoggerService logger2 = LoggerService.getInstance();

        logger1.log("Singleton logger initialized!");
        System.out.println("Are both instances same? " + (logger1 == logger2));
    }
}

Memento Pattern — TextEditor (save/undo)
// TextEditorMemento.java
import java.util.Stack;

class TextMemento {
    private final String state;
    public TextMemento(String state) { this.state = state; }
    public String getState() { return state; }
}

class History {
    private final Stack<TextMemento> stack = new Stack<>();
    public void push(TextMemento m) { stack.push(m); }
    public TextMemento pop() { return stack.isEmpty() ? null : stack.pop(); }
}

class TextEditor {
    private String text = "";
    private final History history;

    public TextEditor(History history) { this.history = history; }

    public void write(String more) { text += more; }
    public String read() { return text; }

    public void save() {
        history.push(new TextMemento(text));
    }

    public void undo() {
        TextMemento m = history.pop();
        if (m != null) this.text = m.getState();
        else System.out.println("No saved state to restore.");
    }
}

public class TextEditorMemento {
    public static void main(String[] args) {
        History history = new History();
        TextEditor editor = new TextEditor(history);

        editor.write("Hello");
        editor.save();
        editor.write(", World");
        editor.save();
        editor.write("!!!");

        System.out.println("Current: " + editor.read());
        editor.undo();
        System.out.println("After undo: " + editor.read());
        editor.undo();
        System.out.println("After second undo: " + editor.read());
    }
}

// OrderProcessorSOLID.java
import java.util.*;

// Domain
class Order {
    public final String id;
    public final List<String> items;
    public final double baseAmount;
    public final String type; // e.g., "STANDARD", "PREMIUM"
    public Order(String id, List<String> items, double baseAmount, String type) {
        this.id = id; this.items = items; this.baseAmount = baseAmount; this.type = type;
    }
}

// Validation (SRP)
interface Validator {
    boolean validate(Order order);
}

class BasicOrderValidator implements Validator {
    public boolean validate(Order order) {
        if (order == null) return false;
        if (order.items == null || order.items.isEmpty()) return false;
        if (order.baseAmount < 0) return false;
        return true;
    }
}

// Pricing (SRP)
interface PriceCalculator {
    double calculatePrice(Order order);
}

class DefaultPriceCalculator implements PriceCalculator {
    public double calculatePrice(Order order) {
        double tax = 0.1 * order.baseAmount;
        double shipping = order.baseAmount > 500 ? 0 : 50;
        return order.baseAmount + tax + shipping;
    }
}

class PremiumPriceCalculator implements PriceCalculator {
    public double calculatePrice(Order order) {
        double tax = 0.08 * order.baseAmount;
        double shipping = 0;
        return order.baseAmount + tax + shipping;
    }
}

// Persistence (SRP/DIP)
interface OrderRepository {
    void save(Order order, double finalAmount);
}

class FileOrderRepository implements OrderRepository {
    public void save(Order order, double finalAmount) {
        System.out.println("Persisting order " + order.id + " with total " + finalAmount + " to file (simulated).");
    }
}

class InMemoryOrderRepository implements OrderRepository {
    private final Map<String, Double> store = new HashMap<>();
    public void save(Order order, double finalAmount) {
        store.put(order.id, finalAmount);
        System.out.println("Saved to memory: " + order.id + "=" + finalAmount);
    }
}

// OrderType Handling (OCP/LSP)
class PriceCalculatorFactory {
    public static PriceCalculator getCalculator(String orderType) {
        if ("PREMIUM".equalsIgnoreCase(orderType)) return new PremiumPriceCalculator();
        return new DefaultPriceCalculator();
    }
}

// OrderProcessor uses dependency injection (DIP)
class OrderProcessor {
    private final Validator validator;
    private final OrderRepository repository;

    public OrderProcessor(Validator validator, OrderRepository repository) {
        this.validator = validator;
        this.repository = repository;
    }

    public void process(Order order) {
        if (!validator.validate(order)) {
            System.out.println("Order validation failed.");
            return;
        }
        PriceCalculator calc = PriceCalculatorFactory.getCalculator(order.type);
        double finalPrice = calc.calculatePrice(order);
        repository.save(order, finalPrice);
        System.out.println("Order processed: " + order.id + " final price: " + finalPrice);
    }
}

public class OrderProcessorSOLID {
    public static void main(String[] args) {
        Order order1 = new Order("O1", Arrays.asList("A", "B"), 400, "STANDARD");
        Order order2 = new Order("O2", Arrays.asList("X"), 1000, "PREMIUM");

        Validator validator = new BasicOrderValidator();
        OrderRepository repo = new FileOrderRepository();

        OrderProcessor processor = new OrderProcessor(validator, repo);
        processor.process(order1);
        processor.process(order2);
    }
}

Observer PatternWeatherStation (Subject) & Displays (Observers)
// WeatherObserverExample.java
import java.util.*;

interface Observer {
    void update(float temp, float humidity, float pressure);
}

interface Subject {
    void registerObserver(Observer o);
    void removeObserver(Observer o);
    void notifyObservers();
}

class WeatherData implements Subject {
    private final List<Observer> observers = new ArrayList<>();
    private float temp, humidity, pressure;

    public void registerObserver(Observer o) { observers.add(o); }
    public void removeObserver(Observer o) { observers.remove(o); }
    public void notifyObservers() {
        for (Observer o : observers) o.update(temp, humidity, pressure);
    }

    public void setMeasurements(float t, float h, float p) {
        this.temp = t; this.humidity = h; this.pressure = p;
        notifyObservers();
    }
}

class CurrentConditionsDisplay implements Observer {
    public void update(float temp, float humidity, float pressure) {
        System.out.println("Current conditions: " + temp + "C and " + humidity + "% humidity.");
    }
}

class StatisticsDisplay implements Observer {
    public void update(float temp, float humidity, float pressure) {
        System.out.println("Stats -> temp: " + temp + ", pressure: " + pressure);
    }
}

public class WeatherObserverExample {
    public static void main(String[] args) {
        WeatherData weatherData = new WeatherData();
        CurrentConditionsDisplay currentDisplay = new CurrentConditionsDisplay();
        StatisticsDisplay statsDisplay = new StatisticsDisplay();

        weatherData.registerObserver(currentDisplay);
        weatherData.registerObserver(statsDisplay);

        weatherData.setMeasurements(25.0f, 65.0f, 1013.0f);
        weatherData.setMeasurements(28.0f, 70.0f, 1012.5f);
    }
}

// RemoteControlCommandExample.java
import java.util.Stack;

// Receiver classes
class Light {
    public void on() { System.out.println("Light is ON"); }
    public void off() { System.out.println("Light is OFF"); }
}

class Fan {
    public void start() { System.out.println("Fan started"); }
    public void stop() { System.out.println("Fan stopped"); }
}

// Command interface
interface Command {
    void execute();
    void undo();
}

// Concrete commands
class TurnOnLightCommand implements Command {
    private final Light light;
    TurnOnLightCommand(Light light) { this.light = light; }
    public void execute() { light.on(); }
    public void undo() { light.off(); }
}

class TurnOffLightCommand implements Command {
    private final Light light;
    TurnOffLightCommand(Light light) { this.light = light; }
    public void execute() { light.off(); }
    public void undo() { light.on(); }
}

class StartFanCommand implements Command {
    private final Fan fan;
    StartFanCommand(Fan fan) { this.fan = fan; }
    public void execute() { fan.start(); }
    public void undo() { fan.stop(); }
}

// Invoker
class RemoteControl {
    private final Stack<Command> history = new Stack<>();

    public void press(Command cmd) {
        cmd.execute();
        history.push(cmd);
    }

    public void undo() {
        if (!history.isEmpty()) {
            Command last = history.pop();
            last.undo();
        } else {
            System.out.println("Nothing to undo.");
        }
    }
}

public class RemoteControlCommandExample {
    public static void main(String[] args) {
        RemoteControl remote = new RemoteControl();
        Light light = new Light();
        Fan fan = new Fan();

        Command lightOn = new TurnOnLightCommand(light);
        Command fanStart = new StartFanCommand(fan);

        remote.press(lightOn);
        remote.press(fanStart);

        System.out.println("Undo last:");
        remote.undo();

        System.out.println("Undo again:");
        remote.undo();
    }
}

        System.out.println("Playing vlc file: " + filename);
    }
}

// Adapter
class MediaAdapter implements MediaPlayer {
    private final AdvancedMediaPlayer advanced;

    public MediaAdapter(AdvancedMediaPlayer advanced) {
        this.advanced = advanced;
    }

    public void play(String type, String filename) {
        if ("mp4".equalsIgnoreCase(type)) advanced.playMp4(filename);
        else if ("vlc".equalsIgnoreCase(type)) advanced.playVlc(filename);
        else System.out.println("Format not supported by adapter: " + type);
    }
}

// Client using new interface but can adapt to old
class AudioPlayer implements MediaPlayer {
    private MediaAdapter adapter;

    public void play(String type, String filename) {
        if ("mp3".equalsIgnoreCase(type)) {
            System.out.println("Playing mp3 file: " + filename);
        } else if ("mp4".equalsIgnoreCase(type) || "vlc".equalsIgnoreCase(type)) {
            adapter = new MediaAdapter(new AdvancedMediaPlayer());
            adapter.play(type, filename);
        } else {
            System.out.println("Invalid media. " + type + " format not supported");
        }
    }
}

public class MediaPlayerAdapterExample {
    public static void main(String[] args) {
        AudioPlayer player = new AudioPlayer();
        player.play("mp3", "song.mp3");
        player.play("mp4", "movie.mp4");
        player.play("vlc", "video.vlc");
        player.play("avi", "my.avi");
    }
}
