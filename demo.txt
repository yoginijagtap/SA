// DesignPatternsDemo.java
// Each pattern implemented simply and runnable in Eclipse

import java.util.*;

// 1. Singleton Pattern - Thread-safe LoggerService
class LoggerService {
    private static volatile LoggerService instance;
    private LoggerService() {}
    public static LoggerService getInstance() {
        if (instance == null) {
            synchronized (LoggerService.class) {
                if (instance == null)
                    instance = new LoggerService();
            }
        }
        return instance;
    }
    public void log(String msg) { System.out.println("Log: " + msg); }
}

// 2. Open/Closed Principle - Shape Area Calculation
abstract class Shape { abstract double area(); }

class Circle extends Shape {
    double r;
    Circle(double r) { this.r = r; }
    double area() { return Math.PI * r * r; }
}

class Square extends Shape {
    double s;
    Square(double s) { this.s = s; }
    double area() { return s * s; }
}

class Triangle extends Shape {
    double b, h;
    Triangle(double b, double h) { this.b = b; this.h = h; }
    double area() { return 0.5 * b * h; }
}

// 3. Single Responsibility Principle - Student Example
class Student {
    String name; int[] marks;
    Student(String name, int[] marks) { this.name = name; this.marks = marks; }
}

class GradeCalculator {
    double average(Student s) {
        return Arrays.stream(s.marks).average().orElse(0);
    }
    String grade(Student s) {
        double avg = average(s);
        return avg >= 90 ? "A" : avg >= 75 ? "B" : "C";
    }
}

class StudentRepository {
    void save(Student s) {
        System.out.println("Saving " + s.name + " to local file...");
    }
}

// 4. Prototype Pattern - Deep Cloning
interface PrototypeShape {
    PrototypeShape cloneShape();
}

class CircleP implements PrototypeShape {
    int r;
    CircleP(int r) { this.r = r; }
    public PrototypeShape cloneShape() { return new CircleP(this.r); }
}

class RectangleP implements PrototypeShape {
    int w, h;
    RectangleP(int w, int h) { this.w = w; this.h = h; }
    public PrototypeShape cloneShape() { return new RectangleP(this.w, this.h); }
}

// 5. Interface Segregation Principle
interface Workable { void work(); }
interface Feedable { void eat(); }
interface Restable { void sleep(); }
interface Payable { void getPaid(); }

class HumanWorker implements Workable, Feedable, Restable, Payable {
    public void work() { System.out.println("Human working"); }
    public void eat() { System.out.println("Human eating"); }
    public void sleep() { System.out.println("Human sleeping"); }
    public void getPaid() { System.out.println("Human getting paid"); }
}

class RobotWorker implements Workable, Payable {
    public void work() { System.out.println("Robot working"); }
    public void getPaid() { System.out.println("Robot maintenance credit"); }
}

// 6. Abstract Factory Pattern
interface MobileCameraFactory {
    PhotoCapturer createPhotoCapturer();
    VideoRecorder createVideoRecorder();
}

interface PhotoCapturer { void capture(); }
interface VideoRecorder { void record(); }

class HighResPhoto implements PhotoCapturer {
    public void capture() { System.out.println("High-res photo captured"); }
}

class HighResVideo implements VideoRecorder {
    public void record() { System.out.println("High-res video recording"); }
}

class StandardPhoto implements PhotoCapturer {
    public void capture() { System.out.println("Standard photo captured"); }
}

class StandardVideo implements VideoRecorder {
    public void record() { System.out.println("Standard video recording"); }
}

class HighResolutionCameraFactory implements MobileCameraFactory {
    public PhotoCapturer createPhotoCapturer() { return new HighResPhoto(); }
    public VideoRecorder createVideoRecorder() { return new HighResVideo(); }
}

class StandardCameraFactory implements MobileCameraFactory {
    public PhotoCapturer createPhotoCapturer() { return new StandardPhoto(); }
    public VideoRecorder createVideoRecorder() { return new StandardVideo(); }
}

// 7. Memento Pattern
class TextEditor {
    private String text = "";
    public void type(String newText) { text += newText; }
    public String getText() { return text; }
    public Memento save() { return new Memento(text); }
    public void undo(Memento m) { text = m.getSavedText(); }

    static class Memento {
        private final String savedText;
        Memento(String text) { savedText = text; }
        String getSavedText() { return savedText; }
    }
}

class History {
    Stack<TextEditor.Memento> history = new Stack<>();
    void save(TextEditor.Memento m) { history.push(m); }
    TextEditor.Memento undo() { return history.pop(); }
}

// 8. Strategy Pattern - Payment
interface PaymentStrategy { void pay(double amount); }

class CreditCardPayment implements PaymentStrategy {
    public void pay(double amt) { System.out.println("Paid " + amt + " via Credit Card"); }
}

class PayPalPayment implements PaymentStrategy {
    public void pay(double amt) { System.out.println("Paid " + amt + " via PayPal"); }
}

class UpiPayment implements PaymentStrategy {
    public void pay(double amt) { System.out.println("Paid " + amt + " via UPI"); }
}

class PaymentContext {
    private PaymentStrategy strategy;
    void setPaymentStrategy(PaymentStrategy s) { strategy = s; }
    void payNow(double amt) { strategy.pay(amt); }
}

// 9. Liskov Substitution Principle Fix
interface Animal { void move(); }

interface Flyable extends Animal { void fly(); }

class BirdL implements Flyable {
    public void move() { System.out.println("Bird walking"); }
    public void fly() { System.out.println("Bird flying"); }
}

class Penguin implements Animal {
    public void move() { System.out.println("Penguin swimming/walking"); }
}

// 10. Proxy Pattern - Image Loading
interface Image { void display(); }

class RealImage implements Image {
    private String filename;
    RealImage(String filename) { this.filename = filename; loadFromDisk(); }
    private void loadFromDisk() { System.out.println("Loading " + filename); }
    public void display() { System.out.println("Displaying " + filename); }
}

class ProxyImage implements Image {
    private RealImage realImage;
    private String filename;
    ProxyImage(String filename) { this.filename = filename; }
    public void display() {
        if (realImage == null) realImage = new RealImage(filename);
        realImage.display();
    }
}

// 11. Iterator Pattern - Menu Example
interface IteratorMenu {
    boolean hasNext();
    String next();
}

class Menu implements IteratorMenu {
    private List<String> items;
    private int pos = 0;
    Menu(String... items) { this.items = Arrays.asList(items); }
    public boolean hasNext() { return pos < items.size(); }
    public String next() { return items.get(pos++); }
}


public class DesignPatternsDemo {
    public static void main(String[] args) {

        System.out.println("\n1. Singleton Pattern:");
        LoggerService.getInstance().log("Application started");

        System.out.println("\n2. Open/Closed Principle:");
        List<Shape> shapes = Arrays.asList(new Circle(2), new Square(3), new Triangle(4,5));
        for (Shape s : shapes) System.out.println("Area: " + s.area());

        System.out.println("\n3. Single Responsibility Principle:");
        Student s = new Student("Swati", new int[]{90, 80, 85});
        GradeCalculator gc = new GradeCalculator();
        System.out.println("Grade: " + gc.grade(s));
        new StudentRepository().save(s);

        System.out.println("\n4. Prototype Pattern:");
        CircleP c1 = new CircleP(5);
        CircleP c2 = (CircleP)c1.cloneShape();
        System.out.println("Cloned circle radius: " + c2.r);

        System.out.println("\n5. Interface Segregation Principle:");
        new HumanWorker().work();
        new RobotWorker().work();

        System.out.println("\n6. Abstract Factory Pattern:");
        MobileCameraFactory factory = new HighResolutionCameraFactory();
        factory.createPhotoCapturer().capture();
        factory.createVideoRecorder().record();

        System.out.println("\n7. Memento Pattern:");
        TextEditor editor = new TextEditor();
        History history = new History();
        editor.type("Hello ");
        history.save(editor.save());
        editor.type("World!");
        System.out.println("Current Text: " + editor.getText());
        editor.undo(history.undo());
        System.out.println("After Undo: " + editor.getText());

        System.out.println("\n8. Strategy Pattern:");
        PaymentContext ctx = new PaymentContext();
        ctx.setPaymentStrategy(new UpiPayment());
        ctx.payNow(500);

        System.out.println("\n9. Liskov Substitution Principle:");
        Animal p = new Penguin();
        p.move();

        System.out.println("\n10. Proxy Pattern:");
        Image img = new ProxyImage("photo.jpg");
        img.display(); // loads
        img.display(); // cached

        System.out.println("\n11. Iterator Pattern:");
        Menu menu = new Menu("Breakfast", "Lunch", "Dinner");
        while (menu.hasNext()) System.out.println(menu.next());
    }
}
